# ======================================================================
# Core crypt() identifiers
# ======================================================================
# MD5
regex:\$1\$[A-Za-z0-9./]{1,16}\$[A-Za-z0-9./]+

# BCRYPT
regex:\$2[abxy]\$\d{2}\$[A-Za-z0-9./]{53}

# SHA-256
regex:\$5\$[$]+\$[A-Za-z0-9./]+

# SHA-512
regex:\$6\$[$]+\$[A-Za-z0-9./]+

# ======================================================================
# Modern/memory-hard hashes
# ======================================================================
# YESCRYPT
regex:\$y\$[$]+\$[$]+\$[A-Za-z0-9./]+

# ARGON2
regex:\$argon2(id|i|d)\$v=\d+\$m=\d+,t=\d+,p=\d+\$[A-Za-z0-9+/=]+\$[A-Za-z0-9+/=]+

# ======================================================================
# LDAP / Directory-style hashes
# ======================================================================
# Generic LDAP HASH Wrapper
regex:\{[A-Z0-9-]+\}[A-Za-z0-9+/=]+

# Salted SHA-1
regex:\{SSHA\}[A-Za-z0-9+/=]+

# Salted SHA-256
regex:\{SSHA256\}[A-Za-z0-9+/=]+

# Salted SHA-512
regex:\{SSHA512\}[A-Za-z0-9+/=]+

# ======================================================================
# PBKDF2 (common formats)
# ======================================================================
# Generic PBKDF2
regex:\$pbkdf2\$[$]+\$[$]+\$[A-Za-z0-9+/=]+

# Broad “what kind of hash is this?” detectors
# Any $id$ crypt-style hash
regex:\$[0-9a-zA-Z]+\$.*

# Any Argon2
regex:\$argon2(id|i|d)\$.*

# Any bcrypt
regex:\$2[abxy]\$.*
